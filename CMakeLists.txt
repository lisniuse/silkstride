cmake_minimum_required(VERSION 3.20)
project(Silkstride VERSION 0.2.2 LANGUAGES CXX)
add_executable(Silkstride WIN32 src/main.cpp)
target_compile_features(Silkstride PRIVATE cxx_std_17)
target_compile_definitions(Silkstride PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(Silkstride PRIVATE gdiplus shell32 shlwapi user32 gdi32 ole32 oleaut32 uuid uiautomationcore advapi32)
if(MSVC)
  set_property(TARGET Silkstride PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(Silkstride PRIVATE /utf-8 /W4)
  target_link_options(Silkstride PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest)
endif()
